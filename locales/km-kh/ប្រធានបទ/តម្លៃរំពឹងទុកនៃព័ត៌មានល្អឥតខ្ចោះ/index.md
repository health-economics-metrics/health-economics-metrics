# តម្លៃរំពឹងទុកនៃព័ត៌មានល្អឥតខ្ចោះ (EVPI)

EVPI គឺជាចំនួនអតិបរមាដែលអ្នកសម្រេចចិត្តគួរបង់ដើម្បីលុបបំបាត់ភាពមិនប្រាកដប្រជាមុនសម្រេច — តម្លៃផ្លូវការនៃ "តោះដំណើរការការសិក្សាមុន"។

## ហេតុអ្វីបានជាសំខាន់

ប្រព័ន្ធសុខាភិបាលប្រឈមនឹងជម្រើសជានិច្ច៖ ទទួលយកឥឡូវលើភស្តុតាងមិនល្អឥតខ្ចោះ ឬផ្តល់មូលនិធិស្រាវជ្រាវបន្ថែមមុន។ EVPI ដាក់លេខលើជម្រើសទីពីរ។ ប្រសិនបើ EVPI គឺ £50,000 ហើយការសាកល្បងដែលបានស្នើចំណាយ £2 លាន ទទួលយកឥឡូវ។ ប្រសិនបើ EVPI គឺ £20 លាន ការសាកល្បងគឺជាការប្រកួតតម្លៃថោក។ សំណួរដូចគ្នា — "តើយើងគួរសាកល្បងនេះមុនពង្រីកទេ?" — កើតឡើងសម្រាប់រាល់ការសម្រេចចិត្តឧបករណ៍សហគ្រាស ហើយស្ទើរតែគ្មាននរណាកំណត់តម្លៃវាទេ។ សម្រាប់កំណត់តម្លៃជម្រើសពង្រីកគម្រោងក្រោយ ជាជាងជម្រើសប្រមូលព័ត៌មានមុន សូមមើល[ការវាយតម្លៃជម្រើសពិត](../ការវាយតម្លៃជម្រើសពិត/)។

## គណិតវិទ្យា

EVPI គឺជាគម្លាតរវាងការសម្រេចចិត្តដោយមានការមើលឃើញជាមុនល្អឥតខ្ចោះ និងការសម្រេចចិត្តឥឡូវលើការរំពឹងទុក៖

```
EVPI = E_θ[ max_j NMB(j, θ) ]  −  max_j E_θ[ NMB(j, θ) ]

θ        = uncertain parameters (with their joint distribution)
NMB(j,θ) = net monetary benefit of option j given θ
```

ពាក្យទីមួយ៖ មធ្យមនៃការទូទាត់ជម្រើសល្អបំផុតឆ្លងពិភពលោកដែលអាចមាននីមួយៗ (អ្នកតែងតែជ្រើសត្រូវ)។ ពាក្យទីពីរ៖ ការទូទាត់នៃជម្រើសតែមួយដែលល្អបំផុតជាមធ្យម (អ្នកត្រូវតែប្តេជ្ញាឥឡូវ)។ EVPI ≥ 0 ជានិច្ច។ EVPI ប្រជាជនគុណនឹងចំនួនការសម្រេចចិត្តដែលរងផលប៉ះពាល់។ គណនាដោយផ្ទាល់ពីការដកយក [PSA](../ការវិភាគភាពរសើបប្រូបាប៊ីលីស្ទិក/)។

## ឧទាហរណ៍ដែលបានគណនា

ដាក់ជំនួយឯកសារ AI ឱ្យប្រើដល់គ្លីនិកករ 5,000 នាក់ ឬមិនដាក់។ ពិភពលោកពីរ៖

```
World A (p = 0.6): assistant saves 20 min/day → NMB of rollout = +£8M
World B (p = 0.4): assistant saves ~0 (workflow friction) → NMB of rollout = −£3M
NMB of "don't roll out" = £0 in both worlds.
```

សម្រេចឥឡូវ៖ E[NMB rollout] = 0.6 × 8 − 0.4 × 3 = **+£3.6M** → ដាក់ឱ្យប្រើ។

ជាមួយព័ត៌មានល្អឥតខ្ចោះ៖ ក្នុងពិភពលោក A ជ្រើសដាក់ឱ្យប្រើ (+£8M) ក្នុងពិភពលោក B ជ្រើសមិនធ្វើអ្វី (£0)។ តម្លៃរំពឹងទុក = 0.6 × 8 + 0.4 × 0 = **£4.8M**។

```
EVPI = 4.8M − 3.6M = £1.2M
```

ការសាកល្បង 3 ខែដ៏តឹងរ៉ឹងដែលចំណាយ £150,000 ដែលដោះស្រាយជាសារវន្តថាអ្នកស្ថិតក្នុងពិភពលោកណា ពិតជាមានតម្លៃយ៉ាងច្បាស់ — ហើយការសាកល្បងណាដែលចំណាយលើស £1.2M មិនមានតម្លៃទេ មិនថាហ្មត់ចត់ប៉ុណ្ណា។

## ទំនាក់ទំនងជាមួយវិស្វកម្មកម្មវិធី

EVPI គឺជាសេដ្ឋកិច្ចនៃ spike ការសាកល្បង A/B និងភស្តុតាងនៃគោលគំនិត។ វាផ្តល់ច្បាប់ជាក់ស្តែងពីរ៖

- **ការសាកល្បងមានតម្លៃផ្តល់មូលនិធិតែបើការសម្រេចចិត្តអាចផ្លាស់ប្តូរពិតប្រាកដ។** ប្រសិនបើអ្នកនឹងដាក់ឱ្យប្រើដោយមិនគិតពីលទ្ធផលសាកល្បង EVPI = 0 ហើយការសាកល្បងគឺជាល្ខោន។
- **កំណត់ការចំណាយសាកល្បងត្រឹម EVPI។** តម្លៃនៃព័ត៌មានត្រូវបានកំណត់ដោយតម្លៃនៃការសម្រេចចិត្តដែលវាផ្តល់ព័ត៌មាន។

EVPI ផ្នែក (EVPPI) ពង្រីកនេះទៅប៉ារ៉ាម៉ែត្រតែមួយ៖ "តើវាមានតម្លៃប៉ុន្មានក្នុងការកំណត់លេខពេលវេលាសន្សំជាក់លាក់?" — ដែលប្រាប់អ្នកពីអ្វីដែលការសាកល្បងគួរវាស់។ សម្រាប់កំណត់តម្លៃការសិក្សា *ជាក់លាក់* ដែលបានស្នើ ជាជាងលុបបំបាត់ភាពមិនប្រាកដប្រជាទាំងអស់ សូមមើល [EVSI](../តម្លៃរំពឹងទុកនៃព័ត៌មានគំរូ/)។

## កំហុសដែលត្រូវជៀសវាង

- **ដំណើរការការសាកល្បងដោយគ្មានច្បាប់សម្រេចចិត្តភ្ជាប់** — ព័ត៌មានដែលមិនអាចផ្លាស់ប្តូរជម្រើសគ្មានតម្លៃតាមនិយមន័យ។
- **មិនយកចិត្តទុកដាក់ចំពោះថ្លៃពន្យារនៃការប្រមូលព័ត៌មាន**៖ ការសាកល្បង 6 ខែពន្យារអត្ថប្រយោជន៍ 6 ខែ ([ថ្លៃដើមនៃការពន្យារ](../ថ្លៃដើមនៃការពន្យារ/)); តម្លៃសុទ្ធនៃការសាកល្បង = EVPI ដែលដោះស្រាយ − ថ្លៃពន្យារ − ថ្លៃសាកល្បង។
- **ចាត់ទុក EVPI ជាការព្យាករណ៍។** វាជាព្រំកំណត់ខាងលើនៃតម្លៃព័ត៌មាន មិនមែនការប៉ាន់ស្មានអ្វីដែលការសិក្សាជាក់លាក់នឹងផ្តល់ទេ។
