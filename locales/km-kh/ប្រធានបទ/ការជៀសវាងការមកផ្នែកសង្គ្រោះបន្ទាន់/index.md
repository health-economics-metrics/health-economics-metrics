# ការជៀសវាងការមកផ្នែកសង្គ្រោះបន្ទាន់

ការជៀសវាងការមកផ្នែកសង្គ្រោះបន្ទាន់រាប់ការមក ED (A&E) និងការចូលសម្រាកសង្គ្រោះបន្ទាន់ដែលបានបង្ការដោយអន្តរាគមន៍ខាងលើ — កម្មវិធីចាត់ថ្នាក់ ការត្រួតពិនិត្យពីចម្ងាយ វួដនិម្មិត ការបង្វែរការថែទាំបន្ទាន់។ វាបម្លែង "យើងចាប់បានឆាប់" ទៅជាការអះអាងដែលបានគិតថ្លៃ។

## ហេតុអ្វីបានជាសំខាន់

ការថែទាំសង្គ្រោះបន្ទាន់គឺជាបរិបទធម្មតាថ្លៃបំផុតក្នុងប្រព័ន្ធ (ថ្លៃដើមឯកតានៃការមក ED ក្នុងចន្លោះ £250–£400 តាមតួលេខ National Cost Collection / PSSRU; ការចូលសម្រាកសង្គ្រោះបន្ទាន់រាប់ពាន់) ហើយការកកស្ទះ ED ធ្លាក់ជាជណ្តើរទៅការពន្យារឡានពេទ្យ និងការវះកាត់តាមគម្រោងដែលលុបចោល។ អ្វីៗដែលដោះស្រាយតម្រូវការខាងលើដោយសុវត្ថិភាព — ដំបូន្មានថែទាំខ្លួនឯង ការថែទាំបឋមថ្ងៃតែមួយ ការឆ្លើយតបសហគមន៍ — ទិញសមត្ថភាពប្រព័ន្ធនៅចំណុចដែលមានភាពតានតឹងបំផុត។ នេះគឺជាបន្ទាត់អត្ថប្រយោជន៍ស្តង់ដារសម្រាប់កម្មវិធីពិនិត្យរោគសញ្ញា សេវាចាត់ថ្នាក់បែប 111 និង[ការត្រួតពិនិត្យអ្នកជំងឺពីចម្ងាយ](../សេដ្ឋកិច្ចនៃការត្រួតពិនិត្យអ្នកជំងឺពីចម្ងាយ/)។

## គណិតវិទ្យា

```
Avoided attendances = population × (baseline rate − intervention rate)
Gross saving        = avoided attendances × unit cost per attendance
                      (+ avoided admissions × admission cost, counted separately)

Net saving          = gross saving − intervention cost − cost of new pathway usage
                      (redirected demand isn't free: a 111 call, a GP slot,
                       a virtual-ward day all have unit costs)
```

ការអះអាងមូលហេតុត្រូវការអ្នកប្រៀបធៀប៖ អត្រាការមកមាននិន្នាការ និងប្រែប្រួលតាមរដូវ ដូច្នេះមុន/ក្រោយតែឯងមិនបង្ហាញអ្វីទេ។

## ឧទាហរណ៍ដែលបានគណនា

សេវាត្រួតពិនិត្យ COPD ពីចម្ងាយសម្រាប់អ្នកជំងឺហានិភ័យខ្ពស់ 3,000 នាក់។ ការវាយតម្លៃក្រុមត្រួតពិនិត្យផ្គូផ្គងបង្ហាញការមក ED ទាក់ទងនឹងការកើតឡើងវិញធ្លាក់ពី 0.9 ទៅ 0.7 ក្នុងមួយអ្នកជំងឺ-ឆ្នាំ និងការចូលសម្រាកសង្គ្រោះបន្ទាន់ពី 0.5 ទៅ 0.42។

```
Avoided attendances = 3,000 × 0.2  = 600 × £300   = £180,000
Avoided admissions  = 3,000 × 0.08 = 240 × £3,800 = £912,000
Gross                                              £1,092,000/year

Costs: monitoring service £600,000; extra community-nurse responses £150,000
Net ≈ +£342,000/year — plus the QALY gains of exacerbations treated earlier.
```

ចំណាំថាបន្ទាត់ការចូលសម្រាកគ្រប់គ្រង៖ ការជៀសវាងការមកតែឯងកម្រចំណាយលើសេវាត្រួតពិនិត្យ; ការជៀសវាង *ការចូលសម្រាក* គឺជាកន្លែងដែលមានប្រាក់។

## ទំនាក់ទំនងជាមួយវិស្វកម្មកម្មវិធី

នេះគឺជា **សេដ្ឋកិច្ចជៀសវាងឧប្បត្តិហេតុ**។ តម្លៃនៃការសង្កេត canary deploys និងប្រព័ន្ធព្រមានមុនគឺជា "ការមកសង្គ្រោះបន្ទាន់" ដែលបានជៀសវាង — ការហៅ បន្ទប់សង្គ្រាម sev-1 — ម្នាក់ៗមានថ្លៃដើមផ្ទុក (ម៉ោងវិស្វករ × អត្រា + ផលប៉ះពាល់អតិថិជន)។ ច្បាប់គំរូដូចគ្នាអនុវត្ត៖ ដកថ្លៃផ្លូវខាងលើថ្មី (ការចាត់ថ្នាក់ការជូនដំណឹងមិនឥតគិតថ្លៃ) ប្រយ័ត្នការជំនួស (ការជូនដំណឹងដែលបង្កើតការងារដោយមិនការពារឧប្បត្តិហេតុគឺជាការថប់បារម្ភសុខភាព មិនមែនសុខភាពទេ) និងបង្ហាញប្រឆាំងការពិតជាមួយក្រុមត្រួតពិនិត្យ (អត្រាឧប្បត្តិហេតុរបស់ក្រុមមាននិន្នាការ និងវិលទៅរកមធ្យម ដូចការមក ED ពិតប្រាកដ)។

## កំហុសដែលត្រូវជៀសវាង

- **ការវិលទៅរកមធ្យម**៖ ក្រុមហានិភ័យខ្ពស់ដែលបានជ្រើសរើសលើឆ្នាំអាក្រក់ប្រសើរឡើងដោយមិនព្យាបាល; ក្រុមត្រួតពិនិត្យផ្គូផ្គង ឬការរចនា stepped-wedge គឺចាំបាច់។
- **តម្រូវការបង្កើតដោយការផ្គត់ផ្គង់**៖ ការចាត់ថ្នាក់ឌីជីថលងាយអាច *បង្កើន* ទំនាក់ទំនងសរុប (កម្រិតទាបជាងដើម្បីស្វែងរកជំនួយ) ខណៈបន្ថយភាគ ED — រាប់ថ្លៃដើមប្រព័ន្ធសរុប។
- **វាយតម្លៃការមកតាមថ្លៃដើមមធ្យម** នៅពេលថ្លៃដើមថេរ ED មិនធ្លាក់ចុះ — សូមមើល[ថ្លៃដើមរឹតបន្តឹង ធៀបនឹងថ្លៃដើមមធ្យម](../ថ្លៃដើមបន្ថែម-ធៀបនឹង-ថ្លៃដើមមធ្យម/)។
