# ការទាញយកឧបយោគតាមការដោះដូរពេលវេលា (TTO)

TTO គឺជាវិធីសាស្ត្រស្តង់ដារសម្រាប់ទាញយកតម្លៃឧបយោគនៃស្ថានភាពសុខភាពដោយផ្ទាល់ពីអ្នកឆ្លើយតប ជាជាងបង្កើតវាឡើង។ វាជាវិធីទាញយកមួយក្នុងចំណោមវិធីផ្សេងទៀត — រួមជាមួយ standard gamble និងការពិសោធជម្រើសដាច់ដោយឡែក — ដែលផលិតសំណុំតម្លៃនៅពីក្រោយឧបករណ៍ដូចជា [EQ-5D](../eq-5d/) ហើយដូច្នេះនៅពីក្រោយការគណនា [QALY](../ឆ្នាំជីវិតដែលកែតម្រូវតាមគុណភាព/) ភាគច្រើនខាងក្រោម។

## ហេតុអ្វីបានជាសំខាន់

រាល់ទម្ងន់ឧបយោគដែលចូលក្នុងការគណនា QALY ត្រូវតែមកពីកន្លែងណាមួយ។ TTO គឺជារបៀប៖ សម្រាប់ស្ថានភាពដែលចាត់ទុកថាប្រសើរជាងការស្លាប់ អ្នកឆ្លើយតបត្រូវបានសួរថាតើប៉ុន្មានឆ្នាំ `X` ក្នុងសុខភាពពេញលេញដែលពួកគេចាត់ទុកថាស្មើនឹង `T` ឆ្នាំក្នុងស្ថានភាពខ្សោយ (`X < T`); ឧបយោគគឺ `X / T`។ សម្រាប់ស្ថានភាពដែលអ្នកឆ្លើយតបខ្លះចាត់ទុកថាអាក្រក់ជាងការស្លាប់ រូបមន្តស្តង់ដារដាច់ (វាមិនអាចតំណាងឧបយោគក្រោមសូន្យបានស្អាតទេ) ដូច្នេះ TTO ពង្រីកត្រូវអនុវត្តជំនួស។ វិស្វករកម្មវិធី ឬអ្នកវិភាគដែលចាត់ទុកទម្ងន់ឧបយោគជាធាតុចូលដែលបានផ្តល់ ដោយមិនដឹងថាវាត្រូវការពិធីការទាញយកដែលបានបញ្ជាក់ត្រឹមត្រូវដើម្បីផលិតវា គឺនៅឆ្ងាយមួយជំហានពីលេខដែលពួកគេមិនអាចការពារបាននៅពេលត្រូវបានប្រកួតប្រជែង។

## គណិតវិទ្យា

```
Standard TTO (state better than death):
  utility = time_in_full_health / time_in_impaired_state

Extended TTO (state worse than death):
  utility = -time_traded_for_death / (total_duration - time_traded_for_death)
```

`time_in_full_health` / `time_in_impaired_state` — ឆ្នាំ `X` ក្នុងសុខភាពពេញលេញដែលវិនិច្ឆ័យថាស្មើនឹង `T` ឆ្នាំក្នុងស្ថានភាពខ្សោយ។ `time_traded_for_death` / `total_duration` — ក្នុងរូបមន្តអាក្រក់ជាងការស្លាប់ ឆ្នាំ `a` នៃជីវិតដែលនៅសល់ `T` ឆ្នាំ ដែលអ្នកឆ្លើយតបនឹងដោះដូរសម្រាប់មរណភាពភ្លាមៗ ដោយចូលចិត្ត `T − a` ឆ្នាំក្នុងសុខភាពពេញលេញរួចស្លាប់ ជាង `T` ឆ្នាំក្នុងស្ថានភាពអាក្រក់ជាងការស្លាប់។ លទ្ធផលអវិជ្ជមាន ចងយ៉ាងនោះថាការស្លាប់ = 0។

## ឧទាហរណ៍ដែលបានគណនា

**ស្តង់ដារ**៖ អ្នកឆ្លើយតបស្ថិតក្នុងស្ថានភាពខ្សោយ 10 ឆ្នាំ ហើយមិនខ្វល់ជាមួយ 7 ឆ្នាំក្នុងសុខភាពពេញលេញ៖ ឧបយោគ = 7 / 10 = **0.7**។

**អាក្រក់ជាងការស្លាប់**៖ ក្នុងជីវិតដែលនៅសល់ 10 ឆ្នាំ អ្នកឆ្លើយតបនឹងដោះដូរ 2 ឆ្នាំសម្រាប់មរណភាពភ្លាមៗ — ពួកគេចូលចិត្ត 8 ឆ្នាំក្នុងសុខភាពពេញលេញរួចស្លាប់ ជាង 10 ឆ្នាំក្នុងស្ថានភាពអាក្រក់ជាងការស្លាប់៖ ឧបយោគ = −2 / (10 − 2) = −2 / 8 = **−0.25**។

## ទំនាក់ទំនងជាមួយវិស្វកម្មកម្មវិធី

ចំណុចដូចគ្នាដែលការស្ទង់មតិ DevEx ឬការចូលរួមជួបប្រទះ នៅពេលសុំឱ្យមនុស្សដាក់ពិន្ទុអ្វីមួយលើមាត្រដ្ឋាន 0–10 ដែលមិនបានពិនិត្យ អនុវត្តនៅទីនេះបញ្ច្រាស់៖ TTO មានពិតជាពីព្រោះ "គ្រាន់តែសុំឱ្យមនុស្សដាក់ពិន្ទុ" មិនមែនជាវិធីទាញយកដែលបានបញ្ជាក់ត្រឹមត្រូវដោយខ្លួនឯងទេ។ មុនពេលសាងសង់សន្ទស្សន៍សមាសធាតុ — ពិន្ទុ DevEx សន្ទស្សន៍ការចូលរួម មាត្រដ្ឋានអស់កម្លាំង — លើលេខដែលដាក់ពិន្ទុដោយខ្លួនឯង សូមសួរថាអ្វីទាញយកវា ហើយតើវិធីនោះត្រូវបានបញ្ជាក់ត្រឹមត្រូវឬទេ ដូចសំណួរដែលអ្នកសេដ្ឋកិច្ចសុខភាពសួរពីទម្ងន់ឧបយោគមុនវាចូលក្នុង QALY។

## ចំណុចងាយធ្លាក់

- **ការធ្វើទូទៅតម្លៃបុគ្គល**៖ តម្លៃ TTO ត្រូវបានទាញយកពី *គំរូ* នៃសាធារណជនទូទៅ (ឬអ្នកជំងឺ) មិនមែនបុគ្គលដែលការថែទាំត្រូវបានសម្រេចទេ — ការប្រើតម្លៃ TTO របស់អ្នកឆ្លើយតបម្នាក់ ដូចជាវាអាចធ្វើទូទៅបាន គឺជាកំហុសគំរូ។
- **រូបមន្តខុសសម្រាប់ស្ថានភាព**៖ រូបមន្ត TTO ស្តង់ដារសន្មត់ថាស្ថានភាពប្រសើរជាងការស្លាប់ដោយគ្មានការស្រពិចស្រពិល; ការអនុវត្តវាលើស្ថានភាពដែលអ្នកឆ្លើយតបខ្លះចាត់ទុកថាអាក្រក់ជាងការស្លាប់ ដោយមិនប្តូរទៅរូបមន្តពង្រីក បង្កើតឧបយោគ (វិជ្ជមាន) ខុសដោយស្ងាត់ស្ងាត់។
- **រយៈពេលមិនអាចប្រៀបធៀបបាន**៖ តម្លៃ TTO ដែលទាញយកដោយប្រើរយៈពេលជីវិតដែលនៅសល់ `T` ខុសគ្នាសម្រាប់ការប្រៀបធៀបអាក្រក់ជាងការស្លាប់ មិនអាចប្រៀបធៀបដោយផ្ទាល់បានទេ ដោយមិនពិនិត្យថាការរចនាការសិក្សារក្សា `T` ថេរ។
