# ឆ្នាំជីវិតដែលកែតម្រូវតាមពិការភាព (DALY)

DALY គឺជាមួយឆ្នាំនៃជីវិតមានសុខភាពល្អដែលបាត់បង់ — កញ្ចក់ផ្នែកបន្ទុកនៃ[QALY](../ឆ្នាំជីវិតដែលកែតម្រូវតាមគុណភាព/)។ ដែល QALYs រាប់សុខភាពដែល *ទទួលបាន* DALYs រាប់សុខភាពដែល *បាត់បង់* ដោយសារជំងឺ; អន្តរាគមន៍ត្រូវបានវាយតម្លៃដោយ DALYs ដែល **បានជៀសវាង**។

## ហេតុអ្វីបានជាសំខាន់

DALY គឺជាស្តង់ដារសុខភាពសកល (WHO ការសិក្សាបន្ទុកជំងឺសកល ហើយក្រសួងសុខាភិបាលប្រទេសចំណូលទាប និងមធ្យមភាគច្រើនធ្វើផែនការជា DALYs)។ ប្រសិនបើកម្មវិធីរបស់អ្នកផ្តោតលើប្រព័ន្ធសុខាភិបាលអន្តរជាតិ ម្ចាស់ជំនួយ ឬកម្មវិធីតម្រឹមតាម WHO ភាសាតម្លៃគឺ DALYs ដែលបានជៀសវាង មិនមែន QALYs ដែលទទួលបានទេ។ ស្តង់ដារប្រវត្តិសាស្ត្រ WHO-CHOICE៖ អន្តរាគមន៍ដែលជៀសវាង DALY មួយក្នុងតម្លៃតិចជាង 1× GDP ក្នុងមួយនាក់ គឺ "មានប្រសិទ្ធភាពថ្លៃដើមខ្ពស់" 1–3× GDP ក្នុងមួយនាក់ "មានប្រសិទ្ធភាពថ្លៃដើម" (ឥឡូវ WHO មិនលើកទឹកចិត្តការប្រើប្រាស់រឹងមាំនៃក្រុមទាំងនេះ ប៉ុន្តែពួកវានៅតែមានគ្រប់ទីកន្លែងក្នុងការអនុវត្ត)។

## គណិតវិទ្យា

```
DALY = YLL + YLD

YLL (years of life lost)         = deaths × standard life expectancy at age of death
YLD (years lived with disability) = prevalence × disability weight

disability weight ∈ [0, 1], 0 = full health, 1 = equivalent to death
(weights published by the Global Burden of Disease study)
```

## ឧទាហរណ៍ដែលបានគណនា

វេទិការំលឹកការត្រួតពិនិត្យក្នុងតំបន់មួយបង្កើនការរកឃើញជំងឺឆាប់។ ជារៀងរាល់ឆ្នាំវាបង្ការការស្លាប់មុនអាយុ 10 (ម្នាក់ៗបាត់បង់ 20 ឆ្នាំធៀបនឹងអាយុកាលស្តង់ដារ) និងបង្ការមនុស្ស 200 នាក់រស់នៅមួយឆ្នាំជាមួយស្ថានភាពដែលទម្ងន់ពិការភាព 0.2។

```
YLL averted = 10 × 20        = 200
YLD averted = 200 × 0.2      = 40
DALYs averted               = 240 per year
```

ប្រសិនបើវេទិកាចំណាយ $600,000/ឆ្នាំដើម្បីដំណើរការ ថ្លៃដើមក្នុងមួយ DALY ដែលបានជៀសវាងគឺ 600,000 / 240 = **$2,500**។ នៅក្នុងប្រទេសដែល GDP ក្នុងមួយនាក់ $8,000 នោះទាបជាងស្តង់ដារ 1× GDP ឆ្ងាយ — "មានប្រសិទ្ធភាពថ្លៃដើមខ្ពស់" ក្នុងន័យ WHO-CHOICE។

## ទំនាក់ទំនងជាមួយវិស្វកម្មកម្មវិធី

- សុខភាពឌីជីថលដែលមានគោលដៅទៅអ្នកផ្តល់មូលនិធិសុខភាពសកល (Gavi, Global Fund, កម្មវិធីជាតិ) គួរបង្ហាញផលប៉ះពាល់ជា **ថ្លៃដើមក្នុងមួយ DALY ដែលបានជៀសវាង** — វាជារង្វាស់ដែលអ្នកពិនិត្យជំនួយគិតរួចហើយ។
- DALY ក៏ជាគំរូ *គណនេយ្យបន្ទុក* មានប្រយោជន៍សម្រាប់វិស្វកម្ម៖ ឧប្បត្តិហេតុ ការបង្កើតមិនស្ថិរភាព និងកកស្ទះកូដចាស់គឺជា "ឆ្នាំរស់នៅជាមួយពិការភាព" សម្រាប់ឃ្លាំងកូដ — បញ្ជីបន្ទុកថ្លឹងតាមការងារធ្ងន់ប្រាប់អ្នកថាកន្លែងណាការជួសជុលទិញ "ឆ្នាំវិស្វកម្មមានសុខភាពល្អ" ច្រើនបំផុត ដូចតារាងបន្ទុក GBD ដឹកនាំចំណាយសុខភាព។

## កំហុសដែលត្រូវជៀសវាង

- **QALYs ដែលទទួលបាន ≠ DALYs ដែលបានជៀសវាងជាលេខ** — ទម្ងន់ខុសគ្នា តារាងជីវិតខុសគ្នា អនុសញ្ញាខុសគ្នា (DALYs ប្រវត្តិសាស្ត្រប្រើការថ្លឹងអាយុ និងការបញ្ចុះតម្លៃក្នុងរង្វាស់)។ កុំបម្លែងដោយធម្មតា។
- **ការប្រើកម្រិតពហុគុណ GDP ជាត្រាបញ្ជាក់** — ខ្លួន WHO ព្រមានថាពួកវាមិនយកចិត្តទុកដាក់ថវិកា និងថ្លៃដើមឱកាស; សូមមើល[កម្រិតឆន្ទៈបង់ប្រាក់](../កម្រិតកំណត់ឆន្ទៈបង់ប្រាក់/)។
- **ការអះអាង DALYs មាត្រដ្ឋានប្រជាជនពីប្រសិទ្ធភាពក្នុងមួយអ្នកប្រើ** ដោយមិនគុណតាមការទទួលយក និងការប្រកាន់ខ្ជាប់ — សូមមើល[ការឈានដល់ និងសមធម៌](../ការឈានដល់-និងសមធម៌/)។
